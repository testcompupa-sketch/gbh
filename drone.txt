#include <Wire.h>
#include <BluetoothSerial.h>
#include <MPU6050_light.h>
#include <ESP32Servo.h>
#include <Adafruit_NeoPixel.h>

// --- Pinout Definitions --- //
#define PIN_ESC1    25  // Rear Right (M1) //
#define PIN_ESC2    26  // Front Right (M2) //
#define PIN_ESC3    32  // Rear Left (M3) //
#define PIN_ESC4    33  // Front Left (M4) // 
#define PIN_NEO_LED  4  // NEO LED Signal Pin //
#define NUM_LEDS     4

// --- Objects & Library Instances --- //
BluetoothSerial SerialBT;
MPU6050 mpu(Wire);
Adafruit_NeoPixel strip(NUM_LEDS, PIN_NEO_LED, NEO_GRB + NEO_KHZ800);

Servo esc1, esc2, esc3, esc4;

// --- Control Command Setpoints --- //
int reqThrottle = 1000; // Range: 1000 to 2000 us //
float reqRoll   = 0;    // Target angle: -30 to +30 degrees //
float reqPitch  = 0;    // Target angle: -30 to +30 degrees //
float reqYaw    = 0;    // Target rate: -180 to +180 deg/s //

// --- PID Gains (Tune these values based on actual flight performance) --- //
float kp_roll = 1.3, ki_roll = 0.04, kd_roll = 18.0;
float kp_pitch = 1.3, ki_pitch = 0.04, kd_pitch = 18.0;

// --- PID State Variables --- //
float error_roll, error_pitch, prev_error_roll = 0, prev_error_pitch = 0;
float pid_i_roll = 0, pid_i_pitch = 0;
float pid_out_roll = 0, pid_out_pitch = 0;

unsigned long prevTime = 0;
bool isArmed = false;

// Function to update NEO LED colors based on armed state //
void updateLEDs(bool armed) {
  for (int i = 0; i < NUM_LEDS; i++) {
    if (armed) {
      strip.setPixelColor(i, strip.Color(0, 255, 0)); // Green = ARMED (Ready to fly) //
    } else {
      strip.setPixelColor(i, strip.Color(255, 0, 0)); // Red = DISARMED (Motors disabled) //
    }
  }
  strip.show();
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22); // SDA = GPIO 21, SCL = GPIO 22 //

  // Initialize Bluetooth Serial //
  SerialBT.begin("ESP32_Drone_PHUP");
  Serial.println("Bluetooth Started! Pair with 'ESP32_Drone_PHUP'");

  // Initialize NEO LEDs //
  strip.begin();
  strip.setBrightness(80);
  updateLEDs(false);

  // Configure ESC PWM timers (50Hz, 1000-2000us pulse width) //
  ESP32PWM::allocateTimer(0);
  ESP32PWM::allocateTimer(1);
  esc1.attach(PIN_ESC1, 1000, 2000);
  esc2.attach(PIN_ESC2, 1000, 2000);
  esc3.attach(PIN_ESC3, 1000, 2000);
  esc4.attach(PIN_ESC4, 1000, 2000);

  // Send minimum throttle pulse to initialize ESCs safely //
  esc1.writeMicroseconds(1000);
  esc2.writeMicroseconds(1000);
  esc3.writeMicroseconds(1000);
  esc4.writeMicroseconds(1000);

  // Initialize and calibrate MPU6050 //
  byte status = mpu.begin();
  if (status != 0) {
    Serial.println("MPU6050 Initialization Failed!");
    while (1);
  }
  Serial.println("Calibrating MPU6050... Keep the drone level and still!");
  delay(1000);
  mpu.calcOffsets(); // Do NOT move the drone during calibration //
  Serial.println("MPU6050 Ready!");
}

void loop() {
  // 1. Read incoming Bluetooth commands //
  if (SerialBT.available()) {
    String command = SerialBT.readStringUntil('\n');
    command.trim();

    // Command structure: "ARM", "DISARM", or "Throttle,Roll,Pitch,Yaw" //
    if (command == "ARM") {
      isArmed = true;
      updateLEDs(true);
    } else if (command == "DISARM") {
      isArmed = false;
      reqThrottle = 1000;
      updateLEDs(false);
    } else {
      // Parse CSV command string (e.g., "1250,5.0,-2.0,0") //
      int firstComma = command.indexOf(',');
      int secondComma = command.indexOf(',', firstComma + 1);
      int thirdComma = command.indexOf(',', secondComma + 1);

      if (firstComma > 0 && secondComma > 0 && thirdComma > 0) {
        reqThrottle = command.substring(0, firstComma).toInt();
        reqRoll     = command.substring(firstComma + 1, secondComma).toFloat();
        reqPitch    = command.substring(secondComma + 1, thirdComma).toFloat();
        reqYaw      = command.substring(thirdComma + 1).toFloat();
      }
    }
  }

  // 2. Fetch current orientation from MPU6050 //
  mpu.update();
  float currentRoll  = mpu.getAngleX();
  float currentPitch = mpu.getAngleY();

  // 3. Compute delta time (dt) //
  unsigned long currentTime = micros();
  float dt = (currentTime - prevTime) / 1000000.0;
  prevTime = currentTime;
  if (dt <= 0 || dt > 0.1) dt = 0.004; // Fallback to 250Hz interval if timing spikes //

  // 4. Compute PID and update ESC outputs //
  if (isArmed && reqThrottle > 1100) {
    // Roll PID Calculation //
    error_roll = reqRoll - currentRoll;
    pid_i_roll += error_roll * dt;
    float pid_d_roll = (error_roll - prev_error_roll) / dt;
    pid_out_roll = (kp_roll * error_roll) + (ki_roll * pid_i_roll) + (kd_roll * pid_d_roll);
    prev_error_roll = error_roll;

    // Pitch PID Calculation //
    error_pitch = reqPitch - currentPitch;
    pid_i_pitch += error_pitch * dt;
    float pid_d_pitch = (error_pitch - prev_error_pitch) / dt;
    pid_out_pitch = (kp_pitch * error_pitch) + (ki_pitch * pid_i_pitch) + (kd_pitch * pid_d_pitch);
    prev_error_pitch = error_pitch;

    // Motor Mixing Algorithm (Quad X Configuration) //
    int m1 = reqThrottle + pid_out_roll - pid_out_pitch; // M1: Rear Right
    int m2 = reqThrottle + pid_out_roll + pid_out_pitch; // M2: Front Right
    int m3 = reqThrottle - pid_out_roll - pid_out_pitch; // M3: Rear Left
    int m4 = reqThrottle - pid_out_roll + pid_out_pitch; // M4: Front Left

    // Constrain pulse signals to standard ESC operating range (1000 - 2000 us) //
    m1 = constrain(m1, 1000, 2000);
    m2 = constrain(m2, 1000, 2000);
    m3 = constrain(m3, 1000, 2000);
    m4 = constrain(m4, 1000, 2000);

    // Send signals to ESCs //
    esc1.writeMicroseconds(m1);
    esc2.writeMicroseconds(m2);
    esc3.writeMicroseconds(m3);
    esc4.writeMicroseconds(m4);
  } else {
    // Safety shutdown: Reset outputs and integral windup when DISARMED or idle //
    esc1.writeMicroseconds(1000);
    esc2.writeMicroseconds(1000);
    esc3.writeMicroseconds(1000);
    esc4.writeMicroseconds(1000);
    pid_i_roll = 0;
    pid_i_pitch = 0;
  }

  delay(4); // Maintain ~250Hz main loop rate //
}
