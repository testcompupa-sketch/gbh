//c++ for controller with drone RX 1.3//
/*
A1-2 : control up and down
A3-5 : contrl forward back left and right
A0 : not used (for more control or sensor)
*/
int move = 0;
int m = 0;
void setup()
{
  //pin input//
  pinMode(2, INPUT);
  pinMode(3, INPUT);
  pinMode(4, INPUT);
  pinMode(5, INPUT);
  pinMode(6, INPUT);
  pinMode(7, INPUT);
  //pin output//
  pinMode(A1, OUTPUT);
  pinMode(A2, OUTPUT);
  pinMode(A3, OUTPUT);
  pinMode(A4, OUTPUT);
  pinMode(A5, OUTPUT);
}

void loop()
{
  //start LOOP reader code//
  if (digitalRead(2) == HIGH) {
    move = 100;
    //up
  }else if (digitalRead(3) == HIGH) {
    move = 101;
    //down
  }else if (digitalRead(4) == HIGH) {
    move = 110;
    //left
  }else if (digitalRead(5) == HIGH) {
    move = 111;
    //right
  }else if (digitalRead(6) == HIGH) {
    m = 01;
  }else if (digitalRead(7) == HIGH) {
    m = 10;
  }
  else{
    move = 000;
    m = 0;
  }
  //end LOOP reader code//
  //processing the code//
  if (move == 110) {

    digitalWrite(A3, HIGH);
    digitalWrite(A4, HIGH);
    digitalWrite(A5, LOW);

  }else if (move == 101) {

    digitalWrite(A3, HIGH);
    digitalWrite(A4, LOW);
    digitalWrite(A5, HIGH);

  }else if (move == 100) {

    digitalWrite(A3, HIGH);
    digitalWrite(A4, LOW);
    digitalWrite(A4, LOW);

  }else if (move == 111) {

    digitalWrite(A3, HIGH);
    digitalWrite(A4, HIGH);
    digitalWrite(A5, HIGH);

  }else if (move == 000) {
    digitalWrite(A3, LOW);
    digitalWrite(A4, LOW);
    digitalWrite(A5, LOW);
  }
  if (m == 01){
    digitalWrite(A1, LOW);
    digitalWrite(A2, HIGH);
  }else if (m == 10){
    digitalWrite(A2, LOW);
    digitalWrite(A1, HIGH);
  }else{
    digitalWrite(A1, LOW);
    digitalWrite(A2, LOW);
  }
  delay(10);
  // delay for stable //
}
